#!/bin/sh
# roblox-external-linux-probe-001 — mirror payload: Void Linux probe (P1)
# Self-contained installer that extracts the Linux scanner snapshot.
#
# Built per mirror/xbps-fetch-mirror-template.txt (sh variant):
#   self-contained — the source snapshot rides inside this file (base64 gzip tar),
#                    so a paste host expiring mid-install can never half-apply it;
#   sha-gated      — the embedded archive sha AND a per-file manifest are verified
#                    before anything is written and again after extraction;
#   fail-closed    — refuses an existing target dir (never overwrites), aborts on
#                    any mismatch, touches nothing outside the dir you pass.
#
# Carries: Linux probe (platform_linux+sober_scanner+port_watcher, tools/sober_probe, CMakeLists, docs)
# Build after extract: sh build_probe.sh  → /tmp/probe ; then flatpak run + /tmp/probe
#
# Usage (run from a file, never from a pipe):
#   sh /absolute/path/to/roblox-external-linux-probe-001.txt /absolute/extract/dir
# The extract dir must NOT exist yet. PASS = the line LINUX_PROBE001_DONE=YES.
# INVERSE (full undo): rm -r /absolute/extract/dir
#
# Fetch via xbps-fetch mirror (paste hosts, anonymous — private repo stays private):
#   xbps-fetch <paste-url> -o /tmp/roblox-external-linux-probe-001.txt   # or: curl -fsSL <paste-url> -o /tmp/f
#   sh /tmp/roblox-external-linux-probe-001.txt /tmp/linux-probe
#   sh /tmp/linux-probe/build_probe.sh
#
set -u
UPDATE=001
SOURCE_COMMIT="1177392"
ARCHIVE_SHA256=59a08f6762dc68a91d4f67cb949b929d943cbfbdaa8eb6b0ebf7083f0d69604b
FILES=14
PAYLOAD_SHA256=c2116f08eb772d5b7df153370b203f96cdb289e2b37608a3978af5aecf23eab9
[ -f "$0" ] || { echo "LINUX_PROBE001_FAIL=run-from-a-file-not-a-pipe (fetch it to a file first, e.g. /tmp/roblox-external-linux-probe-001.txt)"; exit 1; }
[ $# -eq 1 ] || { echo "LINUX_PROBE001_FAIL=usage: sh /absolute/path/to/roblox-external-linux-probe-001.txt /absolute/extract/dir"; exit 1; }
DEST=$1
case "$DEST" in
  /*|[A-Za-z]:/*) ;;
  *) echo "LINUX_PROBE001_FAIL=absolute-path-required got:$DEST (must be absolute, e.g. /tmp/linux-probe)"; exit 1 ;;
esac
if [ -e "$DEST" ]; then
  echo "LINUX_PROBE001_FAIL=target-already-exists:$DEST (never overwrite; rm -r it first or pass a new dir)"
  exit 1
fi
for c in base64 tar sha256sum mktemp; do
  command -v "$c" >/dev/null 2>&1 || { echo "LINUX_PROBE001_FAIL=missing-tool:$c"; exit 1; }
done
TMP=$(mktemp -d) || { echo "LINUX_PROBE001_FAIL=mktemp"; exit 1; }
b64d() { base64 -d "$1" 2>/dev/null || base64 -D "$1" 2>/dev/null; }
sed -n '/^__LINUX_PROBE001_PAYLOAD__$/,$p' "$0" | tail -n +2 > "$TMP/b64.txt"
b64d "$TMP/b64.txt" > "$TMP/src.tar.gz" 2>/dev/null || { echo "LINUX_PROBE001_FAIL=base64-decode"; rm -rf "$TMP"; exit 1; }
GOT=$(sha256sum "$TMP/src.tar.gz" | cut -d' ' -f1)
echo "ARCHIVE_SHA256_GOT=$GOT"
echo "ARCHIVE_SHA256_WANT=$ARCHIVE_SHA256"
if [ "$GOT" != "$ARCHIVE_SHA256" ]; then
  echo "LINUX_PROBE001_FAIL=archive-sha-mismatch (nothing was written; stop and quote both shas)"
  rm -rf "$TMP"
  exit 1
fi
mkdir -p "$DEST" || { echo "LINUX_PROBE001_FAIL=mkdir"; rm -rf "$TMP"; exit 1; }
tar -xzf "$TMP/src.tar.gz" -C "$DEST" || { echo "LINUX_PROBE001_FAIL=tar-extract"; rm -rf "$TMP"; exit 1; }
rm -rf "$TMP"
GOT2=$(cd "$DEST" && find . -type f | sort | sha256sum | cut -d' ' -f1)
echo "PAYLOAD_SHA256_GOT=$GOT2"
if [ "$GOT2" != "$PAYLOAD_SHA256" ]; then
  echo "LINUX_PROBE001_FAIL=payload-sha-mismatch (extraction ok, but content differs)"
  exit 1
fi
echo "LINUX_PROBE001_DONE=YES"
exit 0
