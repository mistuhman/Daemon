std::string TrimLeft(const std::string& s) {
    size_t i = 0;
    while (i < s.size() && std::isspace((unsigned char)s[i])) ++i;
    return s.substr(i);
}

// Parse a single /proc/<pid>/maps line without sscanf.
// Format: "00400000-00409000 r-xp 00000000 08:01 12345 /path"
// We need start, end, perms, path (optional).
bool ParseMapsLine(const std::string& line, uintptr_t& start, uintptr_t& end, std::string& perms, std::string& path) {
    start = 0; end = 0; perms.clear(); path.clear();
    // Find '-' separating start-end
    size_t dash = line.find('-');
    if (dash == std::string::npos) return false;
    size_t sp1 = line.find(' ', dash);
    if (sp1 == std::string::npos) return false;
    std::string s_start = line.substr(0, dash);
    std::string s_end = line.substr(dash + 1, sp1 - dash - 1);
    try {
        start = (uintptr_t)std::stoull(s_start, nullptr, 16);
        end   = (uintptr_t)std::stoull(s_end, nullptr, 16);
    } catch (...) { return false; }
    // perms is next token
    size_t p1 = line.find_first_not_of(' ', sp1);
    if (p1 == std::string::npos) return false;
    size_t p2 = line.find(' ', p1);
    if (p2 == std::string::npos) return false;
    perms = line.substr(p1, p2 - p1);
    // Skip offset, dev, inode (3 tokens)
    size_t cur = p2;
    for (int i = 0; i < 3; ++i) {
        cur = line.find_first_not_of(' ', cur);
        if (cur == std::string::npos) return true; // anon line with no path
        size_t nxt = line.find(' ', cur);
        if (nxt == std::string::npos) return true;
        cur = nxt;
    }
    cur = line.find_first_not_of(' ', cur);
    if (cur != std::string::npos) {
        path = TrimLeft(line.substr(cur));
    }
    return true;
}

} // namespace

// ---------------------------------------------------------------- MapRegions

std::vector<MapRegion> MapRegions(uint32_t pid, bool* denied_out) {
    std::vector<MapRegion> out;
    if (denied_out) *denied_out = false;
    char path[64];
    snprintf(path, sizeof(path), "/proc/%u/maps", pid);
    std::ifstream f(path);
    if (!f) {
        // Permission denied opening maps → yama ptrace_scope (or MAC) is
        // blocking PTRACE_MODE_READ on this pid. Callers must distinguish
        // this from "libroblox not mapped" (which would be a true flatpak-ns
        // or still-starting symptom) so the operator gets the right fix.
        if (denied_out) {
            if (errno == EACCES || errno == EPERM) {
                *denied_out = true;
            } else if (errno == ENOENT || errno == ESRCH) {
                // pid gone — not a permission error, leave false
            }
        }
        return out;
    }
    std::string line;
    while (std::getline(f, line)) {
        uintptr_t start = 0, end = 0;
        std::string perms, mpath;
        if (!ParseMapsLine(line, start, end, perms, mpath)) continue;
        MapRegion r;
        r.start = start;
        r.end = end;
        // maps offset is a separate token; preserve it for ELF header selection.
        std::istringstream tokens(line);
        std::string range, permissions, offset;
        tokens >> range >> permissions >> offset;
        try { r.offset = std::stoull(offset, nullptr, 16); } catch (...) { continue; }
        r.perms = perms;
        r.path = mpath;
        out.push_back(std::move(r));
    }
    return out;
}

// ---------------------------------------------------------------- FindSoberPid

namespace {
uintptr_t VerifiedBase(uint32_t pid, const std::vector<MapRegion>& regions) {
    uintptr_t found = 0;
    for (const auto& region : regions) {
        if (region.offset || region.perms.empty() || region.perms[0] != 'r' ||
            region.path.empty() || region.path[0] != '/') continue;
        // File/memfd labels are not identity. Read the ELF itself, including
        // unnamed/deleted mappings, using exactly the diagnostic's inspector.
        auto image = linux_elf::Inspect(pid, region.start);
        if (!image.valid || image.soname != "libroblox.so") continue;
        bool executable = false;
        for (const auto& code : image.executable_ranges)
            for (const auto& mapping : regions)
                if (mapping.perms.find('x') != std::string::npos &&
                    mapping.start < code.second && mapping.end > code.first)
                    executable = true;
        if (!executable) continue;
        if (found && found != image.bias) return 0; // Ambiguous images: fail closed.
        found = image.bias;
    }
    return found;
}
}

uint32_t FindSoberPid() {
    DIR* directory = opendir("/proc");
    if (!directory) return 0;
    uint32_t found = 0;
    while (auto* entry = readdir(directory)) {
        char* end = nullptr;
        unsigned long value = std::strtoul(entry->d_name, &end, 10);
        if (*end || !value || value > UINT32_MAX) continue;
        const auto pid = static_cast<uint32_t>(value);
        struct stat st{};
        const auto root = "/proc/" + std::to_string(pid);
        if (stat(root.c_str(), &st) || st.st_uid != getuid()) continue;
        // Start-time snapshots prevent accepting an exited/reused process.
        auto start = ProcessStartTicks(pid);
        auto regions = MapRegions(pid);
        if (!VerifiedBase(pid, regions) || start.empty() || start != ProcessStartTicks(pid)) continue;
        if (found && found != pid) { found = 0; break; }
        found = pid;
    }
    closedir(directory);
    return found;
}

uintptr_t LibRobloxBase(uint32_t pid, bool* denied_out) {
    auto start = ProcessStartTicks(pid);
    auto regions = MapRegions(pid, denied_out);
    auto base = VerifiedBase(pid, regions);
    return !start.empty() && start == ProcessStartTicks(pid) ? base : 0;
}

std::string ProcessStartTicks(uint32_t pid) {
    std::ifstream file("/proc/" + std::to_string(pid) + "/stat");
    std::string line; std::getline(file, line);
    auto close = line.rfind(')');
    if (close == std::string::npos) return {};
    std::istringstream fields(line.substr(close + 1));
    std::string value;
    for (int field = 3; field <= 22; ++field) if (!(fields >> value)) return {};
    return value;
}

// ---------------------------------------------------------------- InReadableRegion

bool InReadableRegion(uintptr_t addr, const std::vector<MapRegion>& regions) {
    for (auto& r : regions) {
        if (addr >= r.start && addr < r.end) {
            return !r.perms.empty() && r.perms[0] == 'r';
        }
    }
    return false;
}

// Each field is an absolute address in the target process after Scan().
// taskScheduler is the instance (not the pointer-to-pointer) — i.e. what
// TaskScheduler::Pointer in offsets.h pointed at, resolved via Lea scan.
struct ScanResult {
    uintptr_t taskScheduler = 0;
    uintptr_t renderJob     = 0;
    uintptr_t dataModel     = 0;
    uintptr_t visualEngine  = 0;
    uintptr_t workspace     = 0;
    uintptr_t world         = 0;
    uintptr_t primitives    = 0;

    // Derived: Primitive::* kept for mover::LayoutSane(). Filled after scan
    // by offsets_dynamic::PatchFromScan() if the chain resolved.
    uintptr_t primitiveRotation = 0;
    uintptr_t primitiveFlags    = 0;
    uintptr_t primitivePosition = 0;

    // Human-readable trace — shown on the debug page, photographed for
    // operator receipts. Never empty on failure.
    std::string diag;

    bool ok() const { return taskScheduler && dataModel && workspace; }
};

// One /proc region — parsed from /proc/<pid>/maps. Used to filter
// readable code vs heap vs guard pages before scanning.
struct MapRegion {
    uintptr_t start = 0;
    uintptr_t end = 0;
    uintptr_t offset = 0;
    std::string perms; // e.g. "r-xp", "rw-p", "r--p"
    std::string path;  // may be empty for anon
};

// ---- PID discovery --------------------------------------------------------
// Unique same-user process containing an ELF self-identifying as libroblox.so.
// Reads file/memfd ELF headers; no name scoring or Windows/Wine fallback.
uint32_t FindSoberPid();
std::string ProcessStartTicks(uint32_t pid);

// Verified ELF load bias, not the executable segment start. 0 on missing,
// ambiguous, unreadable or changed process. denied_out reports maps denial.
uintptr_t LibRobloxBase(uint32_t pid, bool* denied_out = nullptr);

// Enumerate /proc/<pid>/maps into regions (perms + path preserved).
// Returns empty on failure. If errno-style feedback is needed, pass a
// non-null denied_out — it is set to true when the maps file could not be
// opened for permission reasons (EACCES/EPERM = yama ptrace_scope blocking
// PTRACE_MODE_READ_ATTACH; ENOENT/ESRCH = pid gone), false on success or
// an unclassifiable IO error.
std::vector<MapRegion> MapRegions(uint32_t pid, bool* denied_out = nullptr);

// Full scan — the only call offsets_dynamic needs. Reads are via
// platform::ReadMemory(pid token, ...) — caller opens the pid once
// and passes the platform handle. This function does not open handles
// itself (keeps /proc permission errors in one place).
ScanResult Scan(void* platform_handle, uint32_t pid, uintptr_t libBase);

// Sober string pooling — mirrors UnknownCheats pooling section.
// Returns empty if not a pooled handle (caller falls back to std::string).
std::string TryPooledString(void* platform_handle, uintptr_t container);

// Utility: validate that an address lies inside a readable rw-p region

// The platform "handle": the Windows API hands out an OS HANDLE; Linux memory
// syscalls take the pid every call, so the token just carries it.
struct GameHandle { uint32_t pid; std::string start; };

std::string ReadFirstLine(const char* path)
{
    FILE* f = fopen(path, "r");
    if (!f) return std::string();
    char buf[512] = {};
    const char* got = fgets(buf, sizeof(buf), f);
    fclose(f);
    return got ? std::string(buf) : std::string();
}

struct Thunk { void (*fn)(void*); void* arg; };

void* PthreadTrampoline(void* p)
{
    Thunk* t = static_cast<Thunk*>(p);
    t->fn(t->arg);
    delete t;
    return nullptr;
}

} // namespace

bool Init()
{
    // No syscall layer to set up on Linux - the memory calls are plain
    // process_vm_readv/writev. XInitThreads must be the first Xlib call of
    // the process when several threads share Xlib (the x11overlay control
    // connection is used from the overlay thread and from feature threads
    // through GameClientRect); SDL's own connection comes later.
#if defined(HAS_X11)
    XInitThreads();
#endif
    return true;
}

uint32_t FindGameProcess()

uintptr_t GameBaseAddress(uint32_t pid)
{
    return sober::LibRobloxBase(pid);

bool OpenGameHandle(uint32_t pid, void** out)
{
    if (!out) return false;
    *out = nullptr;
    const auto start = sober::ProcessStartTicks(pid);
    // Unlike a PID token allocation, verified ELF reads prove read permission.
    if (start.empty() || !sober::LibRobloxBase(pid) || start != sober::ProcessStartTicks(pid)) return false;
    *out = new (std::nothrow) GameHandle{pid, start};
    return *out != nullptr;
}

void CloseHandle(void* h)
{
    delete static_cast<GameHandle*>(h);
}

bool ProcessAlive(void* process_h)
{
    const GameHandle* g = static_cast<GameHandle*>(process_h);
    if (!g) return false;
    return !g->start.empty() && sober::ProcessStartTicks(g->pid) == g->start;
}

bool ReadMem(void* h, uintptr_t addr, void* buf, size_t size)
{
    const GameHandle* g = static_cast<GameHandle*>(h);
    if (!g) return false;
    const iovec local{ buf, size };
    const iovec remote{ (void*)addr, size };
    return process_vm_readv((pid_t)g->pid, &local, 1, &remote, 1, 0) == (ssize_t)size;
}

bool WriteMem(void* h, uintptr_t addr, const void* buf, size_t size)
{
    const GameHandle* g = static_cast<GameHandle*>(h);
    if (!g) return false;
    const iovec local{ const_cast<void*>(buf), size };
    const iovec remote{ (void*)addr, size };
    return process_vm_writev((pid_t)g->pid, &local, 1, &remote, 1, 0) == (ssize_t)size;
}

// ---- clock -------------------------------------------------------------------

uint64_t NowMs()
{
    struct timespec ts{};
    clock_gettime(CLOCK_MONOTONIC, &ts);
    return (uint64_t)ts.tv_sec * 1000ull + (uint64_t)ts.tv_nsec / 1000000ull;
}

uint64_t NowNs()
{
    struct timespec ts{};
    clock_gettime(CLOCK_MONOTONIC, &ts);
    return (uint64_t)ts.tv_sec * 1000000000ull + (uint64_t)ts.tv_nsec;
}

void SleepMs(uint32_t ms)
{
    struct timespec ts{};
    ts.tv_sec = ms / 1000;
    ts.tv_nsec = (long)(ms % 1000) * 1000000l;
    nanosleep(&ts, nullptr);
}

// ---- threads -----------------------------------------------------------------

bool StartThread(void (*fn)(void*), void* arg)
{
    pthread_t t{};
    pthread_attr_t attr{};
    pthread_attr_init(&attr);
    pthread_attr_setdetachstate(&attr, PTHREAD_CREATE_DETACHED);
    Thunk* thunk = new (std::nothrow) Thunk{ fn, arg };
    if (!thunk) { pthread_attr_destroy(&attr); return false; }
    const bool ok = pthread_create(&t, &attr, PthreadTrampoline, thunk) == 0;
    pthread_attr_destroy(&attr);
    if (!ok) delete thunk;
    return ok;
}

void* StartJoinableThread(void (*fn)(void*), void* arg)
{
    pthread_t* t = new (std::nothrow) pthread_t{};
    Thunk* thunk = new (std::nothrow) Thunk{ fn, arg };
    if (!t || !thunk) { delete t; delete thunk; return nullptr; }
    if (pthread_create(t, nullptr, PthreadTrampoline, thunk) != 0) {
        delete t; delete thunk;
        return nullptr;
    }
    return t;
}

bool JoinThread(void* thread_h, uint32_t timeout_ms)
{
    pthread_t* t = static_cast<pthread_t*>(thread_h);
    if (!t) return false;
    struct timespec ts{};
    clock_gettime(CLOCK_REALTIME, &ts);
    ts.tv_sec += timeout_ms / 1000;
    ts.tv_nsec += (long)(timeout_ms % 1000) * 1000000l;
    if (ts.tv_nsec >= 1000000000l) { ts.tv_sec += 1; ts.tv_nsec -= 1000000000l; }
    const bool exited = pthread_timedjoin_np(*t, nullptr, &ts) == 0;
    delete t;
    return exited;
}

void BoostThread()
{
    // Best effort: nicer than Windows' nothing, quieter than SCHED_FIFO.
    sched_yield();
}

void YieldThread()
{
    sched_yield();
}

// ---- files / tools -----------------------------------------------------------

std::string ExeDir()
{
    char buf[4096] = {};
    const ssize_t n = readlink("/proc/self/exe", buf, sizeof(buf) - 1);
    if (n <= 0) return "./";
    const std::string s(buf, (size_t)n);
    const size_t slash = s.find_last_of('/');
    if (slash == std::string::npos) return "./";
    return s.substr(0, slash + 1);
}

std::string CurlPath()
{
    // The transport builds an argument list, so a PATH-resolvable name is
    // what a posix fork/exec version of GetViaCurl wants.
    return "curl";
}

namespace {

bool HasExt(const char* name, const char* ext)
{
    const char* dot = nullptr;
    for (const char* p = name; *p; ++p)
        if (*p == '.') dot = p;
    if (!dot) return false;
    for (const char* a = dot + 1, *b = ext; ; ++a, ++b) {
        char ca = (char)tolower((unsigned char)*a);
        char cb = (char)tolower((unsigned char)*b);
        if (!ca || !cb) return ca == cb;
        if (ca != cb) return false;
    }
}

bool ListFilesWithExt(const char* dir, const char* ext, std::vector<std::string>& out)
{
    out.clear();
    if (!dir || !dir[0] || !ext || !ext[0]) return false;
    mkdir(dir, 0755);   // EEXIST is fine; a bad path fails below

    DIR* d = opendir(dir);
    if (!d) return false;
    const std::string prefix = std::string(dir) + "/";
    while (dirent* e = readdir(d)) {
        if (e->d_name[0] == '.') continue;
        if (!HasExt(e->d_name, ext)) continue;
        struct stat st {};
        if (stat((prefix + e->d_name).c_str(), &st) != 0) continue;
        if (!S_ISREG(st.st_mode)) continue;
        out.push_back(e->d_name);
    }
    closedir(d);
    std::sort(out.begin(), out.end());
    return true;
}

bool OpenUrl(const char* url, std::string& err)
{
    err.clear();
    if (!url || !url[0]) { err = "nothing to open"; return false; }
    // xdg-open returns 0 for "handed off" - unlike ShellExecute there is no
    // "nothing registered" distinction, just the exit code.
    const std::string cmd = std::string("xdg-open '") + url + "' >/dev/null 2>&1";
    if (system(cmd.c_str()) == 0) return true;
    err = "xdg-open could not hand off the link - the link is:\n" + std::string(url);
    return false;
}

// Xlib's default error handler prints and EXITS the process on BadWindow -
// fatal when a window dies between our query and the read (game closing,
// overlay resize). Every call in this layer is best-effort.
int IgnoreXError(Display*, XErrorEvent*) { return 0; }

uint64_t NowMonoMs()
{
    struct timespec ts{};
    clock_gettime(CLOCK_MONOTONIC, &ts);
    return (uint64_t)ts.tv_sec * 1000ull + (uint64_t)ts.tv_nsec / 1000000ull;
}

// Caller holds g_ctl_mutex.
Display* Ctl()
{
    if (!g_ctl) {
        g_ctl = XOpenDisplay(nullptr);
        if (g_ctl) XSetErrorHandler(IgnoreXError);
    }
    return g_ctl;
}

// Format-32 properties come back as arrays of long (32 bits per slot, even on
// LP64) - _NET_CLIENT_LIST, _NET_WM_PID, _NET_ACTIVE_WINDOW all use this.
bool ReadProp32(Display* d, Window w, Atom atom, std::vector<unsigned long>& out)
{
    out.clear();
    if (atom == None) return false;
    Atom type = 0;
    int fmt = 0;
    unsigned long nitems = 0, bytes_after = 0;
    unsigned char* prop = nullptr;
    if (XGetWindowProperty(d, w, atom, 0, 1024, False, AnyPropertyType,
                           &type, &fmt, &nitems, &bytes_after, &prop) != Success || !prop)
        return false;
    const bool ok = (fmt == 32);
    if (ok) {
        const unsigned long* vals = reinterpret_cast<const unsigned long*>(prop);
        out.assign(vals, vals + nitems);
    }
    XFree(prop);
    return ok;
}

std::string ReadPropString(Display* d, Window w, Atom atom)
{
    if (atom == None) return {};
    Atom type = 0;
    int fmt = 0;
    unsigned long nitems = 0, bytes_after = 0;
    unsigned char* prop = nullptr;
    if (XGetWindowProperty(d, w, atom, 0, 1024, False, AnyPropertyType,
                           &type, &fmt, &nitems, &bytes_after, &prop) != Success || !prop)
        return {};
    std::string s(reinterpret_cast<const char*>(prop), nitems);
    XFree(prop);
    return s;
}

std::vector<Window> TopLevelWindows(Display* d, Window root)
{
    std::vector<Window> out;
    Atom list = XInternAtom(d, "_NET_CLIENT_LIST", True);
    std::vector<unsigned long> v;
    if (ReadProp32(d, root, list, v) && !v.empty()) {
        for (unsigned long x : v) out.push_back((Window)x);
        return out;
    }
    Window r = 0, parent = 0, *children = nullptr;
    unsigned n = 0;
    if (XQueryTree(d, root, &r, &parent, &children, &n)) {
        for (unsigned i = 0; i < n; ++i) out.push_back(children[i]);
        if (children) XFree(children);
    }
    return out;
}

bool WindowUsable(Display* d, Window w)
{
    XWindowAttributes wa{};
    return XGetWindowAttributes(d, w, &wa) && wa.map_state == IsViewable &&
           wa.width >= 2 && wa.height >= 2;
}

std::string WindowTitle(Display* d, Window w)
{
    Atom net_name = XInternAtom(d, "_NET_WM_NAME", True);
    std::string t = ReadPropString(d, w, net_name);
    if (t.empty()) t = ReadPropString(d, w, XA_WM_NAME);
    return t;
}

bool TitleIsGame(const std::string& t)
{
    std::string l;
    l.reserve(t.size());
    for (char c : t) l.push_back((char)tolower((unsigned char)c));
    return l.find("roblox") != std::string::npos ||
           l.find("sober") != std::string::npos;
}

// Caller holds g_ctl_mutex.
Window FindGameWindow(Display* d, Window root, uint32_t pid)
{
    const std::vector<Window> wins = TopLevelWindows(d, root);
    if (pid) {
        Atom wm_pid = XInternAtom(d, "_NET_WM_PID", True);
        for (Window w : wins) {
            std::vector<unsigned long> v;
            if (ReadProp32(d, w, wm_pid, v) && !v.empty() && (uint32_t)v[0] == pid)
                return w;
        }
    }
    for (Window w : wins) {
        if (!WindowUsable(d, w)) continue;
        if (TitleIsGame(WindowTitle(d, w))) return w;
    }
    return None;
}

void* HotkeyThread(void*)
{
    // Own Display connection: SDL owns its own, and a passive grab delivers to
    // the grabbing connection. Root is an ancestor of every focus window, so
    // the grab fires no matter who has focus - including our menu, and
    // including the game when the overlay is click-through.
    Display* d = XOpenDisplay(nullptr);
    if (!d) return nullptr;
    XSetErrorHandler(IgnoreXError);
    const Window root = DefaultRootWindow(d);
    const KeyCode k_home = XKeysymToKeycode(d, XK_Home);
    const KeyCode k_f8 = XKeysymToKeycode(d, XK_F8);
    XGrabKey(d, k_home, AnyModifier, root, True, GrabModeAsync, GrabModeAsync);
    XGrabKey(d, k_f8, AnyModifier, root, True, GrabModeAsync, GrabModeAsync);
    XSync(d, False);
    uint64_t last = 0;
    for (;;) {
        XEvent ev;
        XNextEvent(d, &ev);
        if (ev.type != KeyPress || !g_toggle_cb) continue;
        const uint64_t now = NowMonoMs();
        if (now - last < 250) continue;   // auto-repeat + dual delivery debounce
        last = now;
        g_toggle_cb();
    }
    return nullptr;
}

bool AttachSelf(const char* wm_class)
{
    std::lock_guard<std::mutex> lk(g_ctl_mutex);
    Display* d = Ctl();
    if (!d || !wm_class || !wm_class[0]) return false;
    const Window root = DefaultRootWindow(d);
    Atom wclass = XInternAtom(d, "WM_CLASS", True);
    for (Window w : TopLevelWindows(d, root)) {
        std::string cls = ReadPropString(d, w, wclass);
        if (cls.empty()) continue;
        const size_t nul = cls.find('\0');
        const std::string first = cls.substr(0, nul == std::string::npos ? cls.size() : nul);
        bool match = first.size() == strlen(wm_class);
        if (match)
            for (size_t i = 0; i < first.size(); ++i)
                if (tolower((unsigned char)first[i]) != tolower((unsigned char)wm_class[i])) {
                    match = false;
                    break;
                }
        if (!match) continue;
        g_self = w;
        XWindowAttributes wa{};
        if (XGetWindowAttributes(d, g_self, &wa)) g_depth = wa.depth;
        return true;
    }
    return false;
}

bool SetDockHints()
{
    std::lock_guard<std::mutex> lk(g_ctl_mutex);
    Display* d = Ctl();
    if (!d || !g_self) return false;
    const Window root = DefaultRootWindow(d);
    const Atom state = XInternAtom(d, "_NET_WM_STATE", False);
    const Atom above = XInternAtom(d, "_NET_WM_STATE_ABOVE", False);
    const Atom skip_task = XInternAtom(d, "_NET_WM_STATE_SKIP_TASKBAR", False);
    const Atom skip_pager = XInternAtom(d, "_NET_WM_STATE_SKIP_PAGER", False);

    XEvent ev{};
    ev.xclient.type = ClientMessage;
    ev.xclient.window = g_self;
    ev.xclient.message_type = state;
    ev.xclient.format = 32;
    ev.xclient.data.l[0] = 1;                       // _NET_WM_STATE_ADD
    ev.xclient.data.l[1] = (long)above;
    ev.xclient.data.l[2] = (long)skip_task;
    ev.xclient.data.l[3] = 1;                       // source: application
    XSendEvent(d, root, False,
               SubstructureRedirectMask | SubstructureNotifyMask, &ev);
    ev.xclient.data.l[1] = (long)skip_pager;
    ev.xclient.data.l[2] = 0;
    XSendEvent(d, root, False,
               SubstructureRedirectMask | SubstructureNotifyMask, &ev);

    const Atom wtype = XInternAtom(d, "_NET_WM_WINDOW_TYPE", False);
    const Atom utility = XInternAtom(d, "_NET_WM_WINDOW_TYPE_UTILITY", False);
    XChangeProperty(d, g_self, wtype, XA_ATOM, 32, PropModeReplace,
                    reinterpret_cast<const unsigned char*>(&utility), 1);
    XFlush(d);
    return true;
}

bool SetClickThrough(bool enable)
{
    std::lock_guard<std::mutex> lk(g_ctl_mutex);
#if defined(HAS_XFIXES)
    Display* d = Ctl();
    if (!d || !g_self) return false;
    int evb = 0, errb = 0;
    if (!XFixesQueryExtension(d, &evb, &errb)) {
        g_click_through = false;
        return false;
    }
    // XFixesSetWindowShapeRegion with an empty region makes the input shape
    // empty (clicks fall through); a full-window region restores normal input.
    // Shape kind 2 == ShapeInput from <X11/extensions/shape.h> (libXext); the
    // numeric value is protocol-stable and avoids a header dependency on Xext
    // for one constant.
    const int kShapeInput = 2;
    if (enable) {
        XserverRegion empty = XFixesCreateRegion(d, nullptr, 0);
        XFixesSetWindowShapeRegion(d, g_self, kShapeInput, 0, 0, empty);
        XFixesDestroyRegion(d, empty);
    } else {
        XWindowAttributes wa{};
        if (!XGetWindowAttributes(d, g_self, &wa)) return false;
        XRectangle r{};
        r.width = (unsigned short)wa.width;
        r.height = (unsigned short)wa.height;
        XserverRegion full = XFixesCreateRegion(d, &r, 1);
        XFixesSetWindowShapeRegion(d, g_self, kShapeInput, 0, 0, full);
        XFixesDestroyRegion(d, full);
    }
    XFlush(d);
    g_click_through = enable;
    return true;
#else
    (void)enable;
    return false;
#endif
}

bool StartToggleHotkeys(void (*on_toggle)())
{
    {
        std::lock_guard<std::mutex> lk(g_ctl_mutex);
        if (g_hotkeys) return true;
        g_toggle_cb = on_toggle;
    }
    pthread_t t{};
    pthread_attr_t attr{};
    pthread_attr_init(&attr);
    pthread_attr_setdetachstate(&attr, PTHREAD_CREATE_DETACHED);
    const bool ok = pthread_create(&t, &attr, HotkeyThread, nullptr) == 0;
    pthread_attr_destroy(&attr);
    if (ok) {
        std::lock_guard<std::mutex> lk(g_ctl_mutex);
        g_hotkeys = true;
    }
    return ok;
}

bool GameWindowRect(int& x, int& y, int& w, int& h)
{
    std::lock_guard<std::mutex> lk(g_ctl_mutex);
    Display* d = Ctl();
    if (!d) return false;

    static Window cached = None;
    static uint64_t cached_at = 0;
    static int rx = 0, ry = 0, rw = 0, rh = 0;
    const uint64_t now = NowMonoMs();
    if (cached != None && now - cached_at < 100) {
        x = rx; y = ry; w = rw; h = rh;
        return true;
    }

    const Window root = DefaultRootWindow(d);
    const Window gw = FindGameWindow(d, root, g_game_pid);
    if (gw == None) { cached = None; cached_at = now; return false; }

    XWindowAttributes wa{};
    if (!XGetWindowAttributes(d, gw, &wa) || wa.map_state != IsViewable ||
        wa.width < 2 || wa.height < 2) {
        cached = None;
        cached_at = now;
        return false;
    }
    Window child = 0;
    int ax = 0, ay = 0;
    if (!XTranslateCoordinates(d, gw, root, 0, 0, &ax, &ay, &child)) {
        cached = None;
        cached_at = now;
        return false;
    }
    rx = ax; ry = ay; rw = wa.width; rh = wa.height;
    cached = gw;
    cached_at = now;
    x = rx; y = ry; w = rw; h = rh;
    return true;
}

char ActiveWindowKind()
{
    std::lock_guard<std::mutex> lk(g_ctl_mutex);
    Display* d = Ctl();
    if (!d) return 0;
    const Window root = DefaultRootWindow(d);
    Atom active = XInternAtom(d, "_NET_ACTIVE_WINDOW", True);
    std::vector<unsigned long> v;
    if (!ReadProp32(d, root, active, v) || v.empty()) return 0;
    const Window aw = (Window)v[0];
    if (!aw) return 0;
    if (g_self && aw == g_self) return 'u';
    if (g_game_pid) {
        Atom wm_pid = XInternAtom(d, "_NET_WM_PID", True);
        std::vector<unsigned long> pv;
        if (ReadProp32(d, aw, wm_pid, pv) && !pv.empty() && (uint32_t)pv[0] == g_game_pid)
            return 'g';
    }
    return TitleIsGame(WindowTitle(d, aw)) ? 'g' : 'o';
}

static void FeatureLoop() {
    while (!g_request_exit) {
        if (g_base_address && g_layout_ready.load()) {
            features::mover::Tick();
            if (aimbot_enabled) features::RunAimbot();
            features::RunTriggerbot();
            features::RunFlight();
            features::RunGravity();
            features::RunHipHeight();
            features::RunNoFall();
            features::RunFling();
            features::RunFpsCap();
            features::RunJumpPower();
            features::worldenv::RunWorldEnv();
            features::worldesp::RunWorldEsp();
            features::RunAntiafk();
            features::RunNoclip();
            features::RunWalkspeed();
            if (click_teleport_enabled) features::RunClickTeleport();
            features::RunInfiniteJump();
            features::RunFovChanger();
            if (skybox_changer_enabled) features::RunSkyboxChanger();
            features::RunSpectate();
            features::explorer::Run();
            features::RunFreecam();
            features::TickTeleportHold();
            features::servers::Tick();
            if (hitbox_expander_enabled) features::RunHitboxExpander();
        }
        // Idle backoff: while detached every Run* above is skipped by the
        // g_base_address gate, so a 1 ms tick only wakes the CPU ~1000x/s for
        // nothing (the desktop lag reported on Void before Sober attaches).
        // Attach is driven by AttachLoop at 2 Hz, so a 10 Hz tick here changes
        // no attach latency. The ATTACHED cadence is deliberately untouched:
        // the mover's ~600 writes/s depends on the 1 ms tick.
        platform::SleepMs(g_base_address && g_layout_ready.load() ? 1 : 100);
    }
}

static bool RobloxStillAlive() {
    return platform::ProcessAlive(mem::roblox_h.load());
}

static void DetachRoblox() {
    g_layout_ready.store(false);
    features::ReleaseFpsCap();
    features::ReleaseNoFall();
    features::ReleaseFovChanger();
    features::ReleaseFling();
    features::ReleaseJumpPower();
    features::worldenv::ReleaseWorldEnv();
    features::worldesp::ReleaseWorldEsp();
    clear_instance_caches();
    g_base_address = 0;
    mem::process_id.store(0);
    platform::CloseHandle(mem::roblox_h.exchange(nullptr));
    platform::SetGamePid(0);
    g_memory.Handle = nullptr;
    offsets_dynamic::Invalidate();
}

static void AttachLoop(void*) {
    portwatch::ProcessWatcher watcher;
    bool warned_handle = false;

    while (!g_request_exit) {
        if (g_base_address && !RobloxStillAlive()) {
            LogLine("game closed - waiting for it to come back");
            DetachRoblox();
        }

        // Plug-and-play: watcher polls FindSoberPid every tick and reports pid/base change.
        // This enables "launch external before Sober, it auto-attaches" (Fleasion parity).
        uint32_t new_pid = 0;
        uintptr_t new_base = 0;
        if (watcher.Tick(&new_pid, &new_base)) {
            if (new_pid && new_base) {
                LogLine("plug-and-play: sober detected pid %u base 0x%llx — attaching", new_pid, (unsigned long long)new_base);
                if (g_base_address) DetachRoblox();
                mem::process_id.store(new_pid);
                if (mem::grabroblox_h()) {
                    g_base_address = new_base;
                    warned_handle = false;
                    LogLine("attached - pid %u, base 0x%llx (via port_watcher)", new_pid, (unsigned long long)new_base);
                    // Verify read by checking DataModel via scanner
                    if (offsets_dynamic::Ensure()) {
                        g_layout_ready.store(true);
                        LogLine("plug-and-play: %s", offsets_dynamic::diag().c_str());
                        LogLine("watcher: %s", watcher.DebugLine().c_str());
                    } else {
                        LogLine("memory attached; features disabled — %s", offsets_dynamic::diag().c_str());
                    }
                } else {
                    if (!warned_handle) {
                        LogLine("found the game but couldn't open a handle - yama ptrace_scope?");
                        warned_handle = true;
                    }
                    mem::process_id.store(0);
                }
            } else if (!new_pid && g_base_address) {
                LogLine("sober vanished (pid 0) — detaching");
                DetachRoblox();
            }
        }

        // Also watch for game server port change (teleport) when already attached
        if (g_base_address) {
            uint32_t oldp = 0, newp = 0;
            if (watcher.GamePortChanged(&oldp, &newp)) {
                LogLine("portwatch: game port %u -> %u (teleport/server hop) — invalidating world", oldp, newp);
                g_layout_ready.store(false);
                offsets_dynamic::Invalidate();
                // InvalidateWorld is in cache, but we can just log; cache thread will resweep.
            }
        }

        platform::SleepMs(500); // 2 Hz attach poll — heavier than Windows 1s due to plug-and-play want
    }
}

const char* KeyName(int key) {
    if (key == 0) return "unbound";
    if (key == platform::key::LBUTTON) return "mouse 1";
    if (key == platform::key::RBUTTON) return "mouse 2";
    if (key == platform::key::MBUTTON) return "mouse 3";
    if (key == platform::key::MOUSE_X1) return "mouse 4";
    if (key == platform::key::MOUSE_X2) return "mouse 5";
    // Simple static table for common keys
    static char buf[32];
    switch (key) {
        case platform::key::SPACE: return "space";
        case platform::key::ESCAPE: return "escape";
        case platform::key::TAB: return "tab";
        case platform::key::RETURN: return "enter";
        case platform::key::SHIFT: return "shift";
        case platform::key::CONTROL: return "ctrl";
        case platform::key::LSHIFT: return "left shift";
        case platform::key::RSHIFT: return "right shift";
        case platform::key::LCONTROL: return "left ctrl";
        case platform::key::RCONTROL: return "right ctrl";
        case platform::key::LEFT: return "left";
        case platform::key::RIGHT: return "right";
        case platform::key::UP: return "up";
        case platform::key::DOWN: return "down";
        case platform::key::BACK: return "backspace";
        case platform::key::DEL: return "delete";
        default: break;
    }
    if (key >= 'A' && key <= 'Z') {
        buf[0] = (char)key; buf[0] = (char)(buf[0] + 32); buf[1]=0; return buf;
    }
    if (key >= '0' && key <= '9') { buf[0]=(char)key; buf[1]=0; return buf; }
    snprintf(buf, sizeof(buf), "key %d", key);
    return buf;
}

int PollKey() {
    // Linux: poll via platform::SampleKey. De-bounce by waiting for release.
    for (int vk = 1; vk < 256; ++vk) {
        if (vk == platform::key::LBUTTON || vk == platform::key::RBUTTON || vk == platform::key::MBUTTON ||
            vk == platform::key::MOUSE_X1 || vk == platform::key::MOUSE_X2 || vk == platform::key::ESCAPE)
            continue;
        auto ks = platform::SampleKey(vk);
        if (ks.down) {
            // wait for release
            for (int w=0; w<2000; ++w) {
                platform::SleepMs(1);
                if (!platform::SampleKey(vk).down) break;
            }
            return vk;
        }
    }
    static const int buttons[5] = { platform::key::LBUTTON, platform::key::RBUTTON, platform::key::MBUTTON,
                                    platform::key::MOUSE_X1, platform::key::MOUSE_X2 };
    for (int i = 0; i < 5; ++i) {
        if (platform::SampleKey(buttons[i]).down) {
            for (int w=0; w<1000; ++w) {
                platform::SleepMs(1);
                if (!platform::SampleKey(buttons[i]).down) break;
            }
            return buttons[i];
        }
    }
    return 0;
}

// ----------------------------------------------------------------------------
//  what we are running on
// ----------------------------------------------------------------------------
const char* OSName()
{
    static char buf[64];
    if (buf[0]) return buf;
    std::ifstream f("/etc/os-release");
    std::string pretty;
    if (f) {
        std::string line;
        while (std::getline(f, line)) {
            if (line.rfind("PRETTY_NAME=",0)==0) {
                size_t q1 = line.find('"');
                size_t q2 = line.rfind('"');
                if (q1!=std::string::npos && q2!=std::string::npos && q2>q1) pretty=line.substr(q1+1,q2-q1-1);
                else pretty=line.substr(12);
                break;
            }
        }
    }
    if (pretty.empty()) {
        // fallback: /etc/lsb-release or uname
        std::ifstream f2("/etc/lsb-release");
        if (f2) pretty="Linux";
        else pretty="Void Linux";
    }
    snprintf(buf, sizeof(buf), "%s", pretty.c_str());
    return buf;
}

void FillPlayers(menuui::Snapshot& s, const cache::LocalPlayerData& lp)
{
    s.player_count = 0;
    s.spectating   = spectate_enabled;
    s.player_note[0] = '\0';

    if (players_status_msg[0])
        snprintf(s.player_note, sizeof(s.player_note), "%s", players_status_msg);

    const cache::EspSnapshot ents = cache::GetEspSnapshot();
    if (!ents || ents->empty()) {
        if (!s.player_note[0])
            snprintf(s.player_note, sizeof(s.player_note), "%s",
                     s.attached && !s.in_game
                         ? "There is no server to list on the Roblox home page. Join an "
                           "experience and the rows appear here."
                         : "No other players resolved yet. The cache fills itself a few "
                           "seconds after a character spawns.");
        return;
    }

    struct Hit { float dist; const cache::EspEntity* e; };
    Hit hits[128];
    int n = 0;
    for (const cache::EspEntity& e : *ents) {
        if (n >= (int)(sizeof(hits) / sizeof(hits[0]))) break;
        const float dx = e.root_x - lp.x, dy = e.root_y - lp.y, dz = e.root_z - lp.z;
        hits[n].dist = lp.valid ? sqrtf(dx * dx + dy * dy + dz * dz) : 0.0f;
        hits[n].e = &e;
        ++n;
    }
    std::sort(hits, hits + n, [](const Hit& a, const Hit& b) { return a.dist < b.dist; });

    for (int i = 0; i < n && s.player_count < menuui::Snapshot::kMaxPlayers; ++i) {
        const cache::EspEntity& e = *hits[i].e;
        menuui::Snapshot::PlayerRow& r = s.players[s.player_count];

        const char* nm = e.display_name[0] ? e.display_name : e.name;
        snprintf(r.name, sizeof(r.name), "%s", (nm && nm[0]) ? nm : "unknown");
        snprintf(r.account, sizeof(r.account), "%s", e.name[0] ? e.name : r.name);
        r.user_id = e.user_id;
        r.friended = features::friends::IsFriend(e.user_id);

        if (e.IsDead()) {
            snprintf(r.detail, sizeof(r.detail), "dead - %.0f studs", hits[i].dist);
            r.state = 1;
        } else {
            snprintf(r.detail, sizeof(r.detail), "%.0f/%.0f hp - %.0f studs",
                     e.health, e.max_health > 0.0f ? e.max_health : 100.0f, hits[i].dist);
            if (e.tool_name[0]) {
                const size_t len = strlen(r.detail);
                snprintf(r.detail + len, sizeof(r.detail) - len, " - %s", e.tool_name);
            }
            r.state = 0;
        }
        if (spectate_enabled && e.user_id == spectate_user_id) r.state = 2;
        ++s.player_count;
    }
}

// ----------------------------------------------------------------------------
//  debug page: the offset write tests (same logic as Windows, SleepMs)
// ----------------------------------------------------------------------------
void RunTest(int id) {
    const cache::LocalPlayerData& lp = cache::GetLocalPlayer();
    const uintptr_t prim = lp.hrp_primitive;
    if (!is_valid_address(prim)) {
        snprintf(g_snap.test_result, sizeof(g_snap.test_result), "No root part to write to");
        return;
    }

    if (id == 0) {
        const uintptr_t a = prim + Offsets::Primitive::MotionBlock;
        const uintptr_t b = a + Offsets::Primitive::MotionStride;
        float before[18] = {};
        read_raw(a, before, sizeof(float) * 18);
        float blk[18] = {};
        read_raw(a, blk, sizeof(float) * 18);
        blk[10] += 10.0f;
        blk[12] = blk[13] = blk[14] = 0.0f;
        blk[15] = blk[16] = blk[17] = 0.0f;
        const bool w1 = write_raw(a, blk, sizeof(float) * 18);
        const bool w2 = write_raw(b, blk, sizeof(float) * 18);
        float a1[3] = {}, a2[3] = {};
        read_raw(prim + Offsets::Primitive::Position,  a1, sizeof(a1));
        read_raw(prim + Offsets::Primitive::Position2, a2, sizeof(a2));
        snprintf(g_snap.test_result, sizeof(g_snap.test_result),
                 "BLOCK A: %s dY %.2f  |  BLOCK B: %s dY %.2f  (both blocks, whole)",
                 w1 ? "ok" : "FAIL", a1[1] - before[10],
                 w2 ? "ok" : "FAIL", a2[1] - before[10]);
        LogLine("%s", g_snap.test_result);
    } else if (id == 1) {
        const uintptr_t a = prim + Offsets::Primitive::MotionBlock;
        const uintptr_t b = a + Offsets::Primitive::MotionStride;
        float blk[18] = {};
        read_raw(a, blk, sizeof(float) * 18);
        blk[12] = 0.0f; blk[13] = 100.0f; blk[14] = 0.0f;
        const bool wrote = write_raw(a, blk, sizeof(float) * 18) &&
                           write_raw(b, blk, sizeof(float) * 18);
        float v1[3] = {}, v2[3] = {};
        read_raw(prim + Offsets::Primitive::AssemblyLinearVelocity,  v1, sizeof(v1));
        read_raw(prim + Offsets::Primitive::AssemblyLinearVelocity2, v2, sizeof(v2));
        snprintf(g_snap.test_result, sizeof(g_snap.test_result),
                 "VEL: %s  wrote y=100  A reads %.2f, B reads %.2f  %s",
                 wrote ? "ok" : "FAILED", v1[1], v2[1],
                 (v1[1] > 50.0f && v2[1] > 50.0f) ? "LANDED" : "did NOT stick");
        LogLine("%s", g_snap.test_result);
    } else if (id == 4) {
        const uintptr_t a = prim + Offsets::Primitive::Rotation;
        const uintptr_t b = a + (Offsets::Primitive::CFrame2 - Offsets::Primitive::Rotation);
        float p0[3] = {};
        read_raw(prim + Offsets::Primitive::Position, p0, sizeof(p0));
        float blk[18] = {};
        read_raw(a, blk, sizeof(blk));
        const float want_y = p0[1] + 25.0f;
        blk[10] = want_y;
        blk[12] = blk[13] = blk[14] = 0.0f;
        blk[15] = blk[16] = blk[17] = 0.0f;
        write_raw(a, blk, sizeof(blk));
        write_raw(b, blk, sizeof(blk));
        LogLine("--- survival test: +25 studs, one write, sampled 500ms ---");
        LogLine("  target y = %.2f (from %.2f)", want_y, p0[1]);
        char line[160];
        int held = 0, samples = 0;
        for (int t = 0; t <= 500; t += 50) {
            if (t) platform::SleepMs(50);
            float now_a[3] = {}, now_b[3] = {};
            float lin[3] = {};
            read_raw(prim + Offsets::Primitive::Position,  now_a, sizeof(now_a));
            read_raw(prim + Offsets::Primitive::Position2, now_b, sizeof(now_b));
            read_raw(prim + Offsets::Primitive::AssemblyLinearVelocity, lin, sizeof(lin));
            const float dy = now_a[1] - want_y;
            ++samples;
            if (fabsf(dy) < 1.0f) ++held;
            snprintf(line, sizeof(line),
                     "  t+%3dms  A.y %8.2f (off %+7.2f)  B.y %8.2f  vel.y %7.2f",

            LogLine("%s", line);

                 "reverts is the diagnosis (instantly = solver, ~100-300ms = the "
                 "server correcting us, never = the write path is fine).",
                 held, samples);

        const uintptr_t a = prim + Offsets::Primitive::Rotation;
        const uintptr_t b = a + (Offsets::Primitive::CFrame2 - Offsets::Primitive::Rotation);
        const uintptr_t fa = prim + Offsets::Primitive::Flags;
        const uint8_t bit = (uint8_t)Offsets::PrimitiveFlags::Anchored;
        const uint8_t orig = read<uint8_t>(fa);
        float p0[3] = {};
        read_raw(prim + Offsets::Primitive::Position, p0, sizeof(p0));
        LogLine("--- ANCHORED survival test: +25 studs, sampled 500ms ---");
        LogLine("  flags before = 0x%02X (anchored bit %s)",
                (unsigned)orig, (orig & bit) ? "ALREADY SET" : "clear");

        LogLine("  flags after set = 0x%02X  -> anchor %s",
                (unsigned)after_set,
                (after_set & bit) ? "took" : "DID NOT TAKE (write rejected/wrong byte)");
        float blk[18] = {};
        read_raw(a, blk, sizeof(blk));
        const float want_y = p0[1] + 25.0f;
        blk[10] = want_y;
        blk[12] = blk[13] = blk[14] = 0.0f;
        blk[15] = blk[16] = blk[17] = 0.0f;
        write_raw(a, blk, sizeof(blk));
        write_raw(b, blk, sizeof(blk));
        char line[192];

        for (int t = 0; t <= 500; t += 50) {
            if (t) platform::SleepMs(50);
            float na[3] = {}, nb[3] = {}, lin[3] = {};
            read_raw(prim + Offsets::Primitive::Position,  na, sizeof(na));
            read_raw(prim + Offsets::Primitive::Position2, nb, sizeof(nb));
            read_raw(prim + Offsets::Primitive::AssemblyLinearVelocity, lin, sizeof(lin));
            const uint8_t f = read<uint8_t>(fa);
            const float dy = na[1] - want_y;
            ++samples;
            if (fabsf(dy) < 1.0f) ++held;

                     "  t+%3dms  A.y %8.2f (off %+7.2f)  B.y %8.2f  vel.y %7.2f  flags 0x%02X%s",
                     t, na[1], dy, nb[1], lin[1], (unsigned)f,
                     (f & bit) ? "" : "  <- ANCHOR CLEARED");
            LogLine("%s", line);
        }
        write<uint8_t>(fa, orig);
        LogLine("  --- anchor released, flags restored to 0x%02X ---", (unsigned)orig);
        for (int t = 50; t <= 300; t += 50) {
            platform::SleepMs(50);
            float na[3] = {}, lin[3] = {};
            read_raw(prim + Offsets::Primitive::Position, na, sizeof(na));
            read_raw(prim + Offsets::Primitive::AssemblyLinearVelocity, lin, sizeof(lin));
            snprintf(line, sizeof(line),
                     "  post+%3dms  A.y %8.2f (rel start %+7.2f)  vel.y %7.2f",
                     t, na[1], na[1] - want_y, lin[1]);
            LogLine("%s", line);
        }
        LogLine("--- anchored survival test end ---");
        snprintf(g_snap.test_result, sizeof(g_snap.test_result),
                 "Anchored: held %d/%d over 500ms. If the anchor bit shows CLEARED, "
                 "the flags write is being undone. If it held but y still fell, the "
                 "anchor does not own position here. See the log.",
                 held, samples);
    } else {
        LogLine("--- primitive float probe (hrp 0x%llX) ---", (unsigned long long)prim);
        char line[128];
        for (int off = 0xA0; off <= 0x1C0; off += 0x10) {
            float vals[4] = {};

            LogLine("%s", line);
        }
        snprintf(g_snap.test_result, sizeof(g_snap.test_result),
                 "Probe written to the log below (0xA0..0x1C0)");
        LogLine("--- probe end ---");
    }
}

Status Poll()
{
    Status s;

    s.sober_pid = sober::FindSoberPid();
    s.sober = s.sober_pid != 0;
    s.libroblox = false;
    s.maps_denied = false;
    if (s.sober) {
        // Distinguish three states so the operator gets the right fix line:
        //   (a) maps readable + libroblox found  → ready to load
        //   (b) maps UNREADABLE (EACCES)         → yama ptrace_scope problem
        //   (c) maps readable, libroblox missing → true flatpak-ns / early start

        s.libroblox = (base != 0);
        s.maps_denied = denied && base == 0;

        if (r == g_overlay_pid || (r < 0 && errno == ECHILD)) g_overlay_pid = -1;

    //  Sober -> memory access (yama — gate this BEFORE libroblox because a
    //  yama block causes libroblox to appear "missing" too) -> display -> binary.
    // Successful cross-process ELF reads are stronger evidence than a global
    // policy value (a permitted process may read even with ptrace_scope > 0).
    s.ready = s.sober && s.libroblox && s.display && s.overlay_file;

    if (s.overlay_running) {
        snprintf(s.line, sizeof(s.line), "overlay running (pid %d)", (int)g_overlay_pid);
        snprintf(s.detail, sizeof(s.detail), "locked to Sober — stop it here or close Sober");
    } else if (!s.sober) {
        snprintf(s.line, sizeof(s.line), "looking for Sober...");
        if (s.yama > 0) {
            snprintf(s.detail, sizeof(s.detail),

        } else {
            snprintf(s.detail, sizeof(s.detail),
                     "no unique readable libroblox ELF; open a game or inspect permissions/multiple instances with the diagnostic");
        }

        // yama blocks PTRACE_MODE_READ on /proc/<pid>/maps — this is the #1
        // reason libroblox "isn't visible", and it's the fix the operator can
        // apply without reinstalling anything. Show it BEFORE the flatpak-ns
        // guess so the "run as root" suggestion doesn't mask it.
        snprintf(s.line, sizeof(s.line), "Sober found (pid %u) — not ready", s.sober_pid);

                 "inspect target credentials and security policy.", s.sober_pid);

        // flatpak-ns / bwrap-startup case.
        snprintf(s.line, sizeof(s.line), "Sober found (pid %u) — not ready", s.sober_pid);
        snprintf(s.detail, sizeof(s.detail),

bool StartOverlay(char* err, std::size_t errlen)
{
    if (err && errlen) err[0] = '\0';
    if (g_overlay_pid > 0) {
        if (err && errlen) snprintf(err, errlen, "overlay already running (pid %d)", (int)g_overlay_pid);
        return false;
    }
    const std::string exe = OverlayPath();
    if (!FileExists(exe)) {
        if (err && errlen) snprintf(err, errlen, "no overlay binary at %s", exe.c_str());
        return false;
    }
    const pid_t pid = fork();
    if (pid < 0) {
        if (err && errlen) snprintf(err, errlen, "fork failed (errno %d)", errno);
        return false;
    }
    if (pid == 0) {
        // Child: the overlay in locked mode — hidden unless Sober's window is
        // there, geometry following it. Its own process group so our Ctrl-C
        // does not race it; it exits on SIGTERM with its restore path.
        setsid();
        const char* argv[] = { "external", "--locked", nullptr };
        execv(exe.c_str(), (char* const*)argv);
        _exit(127);
    }
    g_overlay_pid = pid;
    return true;
}

bool StopOverlay()
{
    if (g_overlay_pid <= 0) return false;
    kill(g_overlay_pid, SIGTERM);
    for (int i = 0; i < 30; ++i) {
        int st = 0;
        if (waitpid(g_overlay_pid, &st, WNOHANG) == g_overlay_pid) {
            g_overlay_pid = -1;
            return true;
        }
        usleep(100 * 1000);
    }
    // Not exiting after 3 s: leave it, report the pid; its writes restore only
    // on a clean exit, so we do NOT SIGKILL by default.
    return true;
}

#pragma once
// Linux ELF inspection shared by diagnostics and attachment. No game offsets.
#ifdef __linux__
#include <algorithm>
#include <cstdint>
#include <cstring>
#include <elf.h>
#include <string>
#include <vector>
#include <utility>
#include <sys/uio.h>
#include <unistd.h>
namespace linux_elf {
struct Image {
    uintptr_t bias = 0;
    bool valid = false;
    std::string soname;
    std::string build_id;
    std::vector<std::pair<uintptr_t, uintptr_t>> executable_ranges;
};
inline bool Read(unsigned pid, uintptr_t address, void* out, size_t size) {
    if (size > UINTPTR_MAX - address) return false;
    iovec local{out, size}, remote{reinterpret_cast<void*>(address), size};
    return process_vm_readv(pid, &local, 1, &remote, 1, 0) == static_cast<ssize_t>(size);
}
inline Image Inspect(unsigned pid, uintptr_t headerAddress) {
    Image image;
    Elf64_Ehdr header{};
    if (!Read(pid, headerAddress, &header, sizeof(header)) ||
        std::memcmp(header.e_ident, ELFMAG, SELFMAG) ||
        header.e_ident[EI_CLASS] != ELFCLASS64 || header.e_ident[EI_DATA] != ELFDATA2LSB ||
        header.e_ident[EI_VERSION] != EV_CURRENT || header.e_version != EV_CURRENT ||
        (header.e_type != ET_DYN && header.e_type != ET_EXEC) ||
        header.e_phentsize != sizeof(Elf64_Phdr) || !header.e_phnum || header.e_phnum > 256 ||
        header.e_phoff > 1048576 || header.e_phoff > UINTPTR_MAX - headerAddress) return image;
    std::vector<Elf64_Phdr> ph(header.e_phnum);
    if (!Read(pid, headerAddress + header.e_phoff, ph.data(), ph.size() * sizeof(Elf64_Phdr))) return image;
    for (const auto& p : ph) {
        if (p.p_type != PT_LOAD) continue;
        if (p.p_filesz > p.p_memsz || p.p_vaddr > UINTPTR_MAX - p.p_memsz) return image;
        if (p.p_offset == 0 && p.p_filesz >= sizeof(header) && p.p_vaddr <= headerAddress) {
            if (image.valid && image.bias != headerAddress - p.p_vaddr) return Image{};
            image.bias = headerAddress - p.p_vaddr;
            image.valid = true;
        }
    }
    if (!image.valid) return image;
    for (const auto& p : ph) {
        if (p.p_type != PT_LOAD || !(p.p_flags & PF_X)) continue;
        if (p.p_vaddr > UINTPTR_MAX - image.bias || p.p_memsz > UINTPTR_MAX - (image.bias + p.p_vaddr)) return Image{};
        image.executable_ranges.emplace_back(image.bias + p.p_vaddr, image.bias + p.p_vaddr + p.p_memsz);
    }
    auto inside = [&](uintptr_t address, size_t size) {
        for (const auto& p : ph) {
            if (p.p_type != PT_LOAD || p.p_vaddr > UINTPTR_MAX - image.bias) continue;
            auto begin = image.bias + p.p_vaddr;
            if (address >= begin && address - begin <= p.p_memsz &&
                size <= p.p_memsz - (address - begin) && size <= UINTPTR_MAX - address) return true;
        }
        return false;
    };
    auto read = [&](uintptr_t address, void* out, size_t size) {
        return inside(address, size) && Read(pid, address, out, size);
    };
    for (const auto& p : ph) {
        if (p.p_type != PT_DYNAMIC || p.p_vaddr > UINTPTR_MAX - image.bias) continue;
        auto dynamicAddress = image.bias + p.p_vaddr;
        uintptr_t strings = 0;
        size_t length = 0, nameOffset = 0;
        bool hasName = false, terminated = false;
        for (size_t i = 0; i < std::min<size_t>(p.p_memsz / sizeof(Elf64_Dyn), 4096); ++i) {
            Elf64_Dyn dyn{};
            auto offset = i * sizeof(dyn);
            if (offset > UINTPTR_MAX - dynamicAddress || !read(dynamicAddress + offset, &dyn, sizeof(dyn))) break;
            if (dyn.d_tag == DT_NULL) { terminated = true; break; }
            if (dyn.d_tag == DT_STRTAB) strings = dyn.d_un.d_ptr;
            if (dyn.d_tag == DT_STRSZ) length = dyn.d_un.d_val;
            if (dyn.d_tag == DT_SONAME) { nameOffset = dyn.d_un.d_val; hasName = true; }
        }
        if (!terminated || !hasName || !strings || nameOffset >= length || length > 64 * 1024 * 1024) continue;
        std::vector<uintptr_t> choices;
        if (inside(strings, length)) choices.push_back(strings);
        if (strings <= UINTPTR_MAX - image.bias && inside(image.bias + strings, length) &&
            (choices.empty() || choices[0] != image.bias + strings)) choices.push_back(image.bias + strings);
        if (choices.size() != 1) continue; // Never choose an ambiguous pointer representation.
        std::string name;
        for (size_t i = 0; i < std::min<size_t>(length - nameOffset, 512); ++i) {
            char ch;
            if (!read(choices[0] + nameOffset + i, &ch, 1)) break;
            if (!ch) { image.soname = name; break; }
            if (static_cast<unsigned char>(ch) < 32 || static_cast<unsigned char>(ch) == 127) break;
            name += ch;
        }
    }
    for (const auto& p : ph) {
        if (p.p_type != PT_NOTE || !p.p_filesz || p.p_filesz > 65536 || p.p_vaddr > UINTPTR_MAX - image.bias) continue;
        std::vector<unsigned char> notes(p.p_filesz);
        if (!read(image.bias + p.p_vaddr, notes.data(), notes.size())) continue;
        size_t at = 0;
        while (at + sizeof(Elf64_Nhdr) <= notes.size()) {
            Elf64_Nhdr note;
            std::memcpy(&note, notes.data() + at, sizeof(note));
            at += sizeof(note);
            size_t names = (size_t(note.n_namesz) + 3) & ~size_t(3);
            size_t desc = (size_t(note.n_descsz) + 3) & ~size_t(3);
            if (names > notes.size() - at || desc > notes.size() - at - names) break;
            if (note.n_type == NT_GNU_BUILD_ID && note.n_namesz == 4 &&
                !std::memcmp(notes.data() + at, "GNU", 4)) {
                static const char hex[] = "0123456789abcdef";
                for (size_t i = 0; i < note.n_descsz; ++i) {
                    auto c = notes[at + names + i];
                    image.build_id += hex[c >> 4]; image.build_id += hex[c & 15];
                }
            }
            at += names + desc;
        }
    }
    return image;
}
